File -> List[TYPE=ModuleRequest]
        List[TYPE=StaticImport]
        List[TYPE=]
        List[TYPE=StarExport]
        (BBContainer)+
      | (BB)+

BBContainer -> Bindings
               List[TYPE=BB]

Bindings -> List[TYPE=EnvBinding]
            List[TYPE=RemoteEnvBinding]
            List[TYPE=PoolBinding]

BB -> (STMT)+

EnvBinding
  - string : NAME
  - bool   : ASW                           <- Probably Useless
    - bool   : JSARG
    - bool   : JSRESTARG
    - bool   : JSLET
    - bool   : JSCONST
    - bool   : JSVAR
  - bool*  : LINK
  - double : IDX                           <- Probably Useless
  - double : REFIDX
  - double : Scope
  - double : ParentScope                   <- Probably Useless
  - double : NEXT

RemoteEnvBinding : [EnvBinding]
  - bool   : NSIMPORT
  - bool   : LINK
  - bool   : MODULETOPLEVELBINDING                   <- Probably Useless
  - double : REFIDX

STMT := JSExplicitBindingDeclarationN     {Transitional}
      | JSExplicitBindingDeclaration      {Transitional}
      | JSFuncDeclSEXP                    {Transitional}
      | SiblingSpecialWrite               {Transitional}
      | ResolveContinueTarget             {Transitional}
      | ResolveBreakTarget                {Transitional}
      | JSImplicitBindingDeclaration      {Transitional}

      <-- Env OPs -->
      | LWrite
        - XWrite  - Non Aliased binding local stack binding
        - LWrite  - Aliased local stack binding
      | RWrite
        - XRWrite - Exclusive ownership over remote binding
        - RWrite  - Non exclusive owenership over remote binding
      | FieldWrite
        - LFWrite
        - RFWrite
        - JSGFWrite
      | GWrite
        - JSGWrite
      | JSSloppyFuncDecl + JSSloppyDecl
        - JSGDecl
      | JSSuperFieldWrite
        - JSSFWrite
      | JSComputedFieldWrite
        - JSCFWrite
      | JSPrivateFieldWrite
        - JSPFWrite

      <-- CFlow -->
      | ReturnAsync
      | Return
      | IfJump                            {Only-Execution-Target}
      | IfElseJump
      | Goto

      <-- Async CFlow -->
      | JSInitialYield
      | Yield

      <-- Err Handling -->
      | PushCatchContext
      | Throw
      | PopCatchContext
      | InvokeFinalizer
      | Ret
      | PopFinalizerReturnTarget

      <-- Stack OPs -->
      | StackReject
      | StackRetain

      <-- Loop Related -->
      | LoopInitPreludeEnd

      <-- Garbage -->
      | NOP

EnvAtoms := EnvBinding
          | RemoteEnvBinding
          | GlobalBinding

RVAL := ResolveEnvBinding                 {Transitional}
      | ResolvePrivateEnvBinding          {Transitional}
      | PVTEnvRead                        {Transitional}
      | TDZRead                           {Transitional}

      <-- CFlow Transfer -->
      | CallSite
      | Apply
      | Await

      <-- Environment OPs -->
      | EnvRead
        - SAFE
        - TAINTED

      <-- OBJ OPs -->
      | FieldRead
      | JSPrivateFieldRead
      | JSComputedFieldRead
      | JSSuperFieldRead

      <-- Primitives -->
      | String
      | Lambda                            {Semi-Transitional}
      | Number
      | PoolBinding
      | Boolean
      | Null
      | RegExp
      | JSNUBD
      | JSClass
      | JSPrivate
      | JSObject
      | JSArray
      | JSTemplate
      | BitInt                            <- Typo here

      <-- VSysCall [Proto] -->
      | JSSetPrototypeOf
      | JSCheckConstructor
      | JSSetHome
      | JSADDBRAND

      <-- VSysCall [Object] -->
      | JSToObject
      | JSDefineObjProp
      | JSDefineObjMethod
      | JSCopyDataProperties

      <-- VSysCall [Array] -->
      | JSAppend

      <-- VSysCall [Methods] -->
      | JSSetName

      <-- STKOPs -->
      | StackPop                          <- Not very informative

      <-- Binops -->
      | Unop
      | Binop
      | JSBinop
      | JSUnop
        | UNOPDelMemberExpr
        | UNOPDelVar
      | IDOP
      | JSIDOP

      | JSForOfIteratorClose              <- This has no business being an RVAL
      | JSForInStart                      <- Messy RVAL
      | JSForInNext                       <- Messy RVAL
      | JSForOfStart                      <- Messy RVAL
      | JSForOfNext                       <- Messy RVAL
      | JSCatchContext                    <- This is just an StackPop, merge + flags maybe??
      | StackToHeap                       <- This has no business being an RVAL

      | JSCTX                             <- VSysCall
